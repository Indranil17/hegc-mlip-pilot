cmake_minimum_required(VERSION 3.18)
project(hegc_cxx LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# Stand-alone C++ tests (no Python needed)
enable_testing()
add_executable(hegc_core_tests tests/test_core.cpp)
target_include_directories(hegc_core_tests PRIVATE include)
target_compile_options(hegc_core_tests PRIVATE -Wall -Wextra)
find_package(Threads REQUIRED)
add_test(NAME core COMMAND hegc_core_tests)
